# 群机器人「发图」按需截图 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 群成员在企业微信群发「机器人发图」时,服务端记录该群所属商家的发图请求,采集端轮询后对该账号截图并发回该群。

**Architecture:** 服务端(xc_store)新增 WorkTool 回调接口(记录请求进内存缓存)+ 轮询/回报接口;采集端(pinduoduo_mc)在商家操作线程主循环里轮询,复用现有单账号「截图→COS→发群」能力。服务端只记录、采集端异步发图,契合 WorkTool 回调不支持同步下发指令的限制。

**Tech Stack:** 服务端 Next.js 14 (Pages Router) + TypeScript + SQLite(现有);采集端 Python + PySide6 + Playwright + aiohttp(现有)。**不引入任何新依赖。**

## Global Constraints

- **不引入新依赖 / 新框架**(用户明确要求)。验证方式:服务端用 `npx tsc --noEmit` 类型检查 + `curl` 手动验证;采集端用真实测试群 + 运行日志验证。无自动化测试框架。
- 回调请求体字段(WorkTool 契约,verbatim):`spoken`(文本)、`groupName`(群名)、`roomType`(1=外部群,2=外部联系人,3=内部群,4=内部联系人)、`textType`(1=文本,2=图片,…)。回调**必须**返回 `{ "code": 0, "message": "success" }`。
- 触发条件:`roomType ∈ {1,3}` 且 `textType === 1` 且 `String(spoken).trim() === '机器人发图'`(精确匹配)。
- 群名匹配:`merchant.groupName === body.groupName`(精确)。
- 缓存生命周期:`pending` →(采集端拉取)`processing` →(采集端回报成功)移除;`PROCESSING_TTL = 5min` 退回 pending,`REQUEST_TTL = 10min` 作废。
- 回调鉴权:URL query `?token=`,比对 `process.env.WORKTOOL_CALLBACK_TOKEN`。轮询/回报鉴权:`x-api-key` 比对 `db.getSetting('apiKey')`(现有 public 范式)。
- 采集端账号匹配:`account_data["api_id"] === merchantId`;发群目标:`account_data["groupName"]`。
- git 提交:每个 Task 末尾的 commit 命令仅在**用户明确要求提交时**执行;否则完成实现即可,提交交给用户。commit message 不带任何 AI 署名。

- [ ] **Step 1: 写缓存模块**

Create `xc_store/src/lib/screenshot-request-cache.ts`:

```ts
import { Merchant } from '@/lib/types';

// 「机器人发图」请求的内存缓存。
// 生命周期：pending →(采集端拉取)processing →(采集端回报成功)移除。
// 兜底：processing 超 PROCESSING_TTL 退回 pending 重试；pending 超 REQUEST_TTL 作废。

export type RequestStatus = 'pending' | 'processing';

export interface ScreenshotRequest {
  merchantId: string;   // merchant.id（采集端 account_data.api_id）
  merchantName: string; // merchant.name
  groupName: string;    // merchant.groupName（缓存 key）
  status: RequestStatus;
  requestedAt: number;  // 首次入队时间戳(ms)
  updatedAt: number;    // 最近更新时间戳(ms)
}

const PROCESSING_TTL = 5 * 60 * 1000;  // processing 超 5min 退回 pending
const REQUEST_TTL = 10 * 60 * 1000;    // pending 超 10min 作废

// 挂在 globalThis 上，避免 Next.js dev 热重载丢失；生产单进程 fork 天然单例。
const globalForCache = globalThis as unknown as {
  __screenshotRequestCache__?: Map<string, ScreenshotRequest>;
};

function store(): Map<string, ScreenshotRequest> {
  if (!globalForCache.__screenshotRequestCache__) {
    globalForCache.__screenshotRequestCache__ = new Map<string, ScreenshotRequest>();
  }
  return globalForCache.__screenshotRequestCache__;
}

// 每次读写前调用：处理 TTL 兜底。
function sweep(now: number): void {
  const map = store();
  for (const [groupName, req] of map) {
    if (req.status === 'processing' && now - req.updatedAt > PROCESSING_TTL) {
      req.status = 'pending';
      req.updatedAt = now;
    } else if (req.status === 'pending' && now - req.requestedAt > REQUEST_TTL) {
      map.delete(groupName);
    }
  }
}

// 入队。同群已存在(pending/processing)时只刷新 updatedAt（合并去重）。
export function enqueue(merchant: Merchant): void {
  const now = Date.now();
  sweep(now);
  const map = store();
  const existing = map.get(merchant.groupName);
  if (existing) {
    existing.updatedAt = now;
    return;
  }
  map.set(merchant.groupName, {
    merchantId: merchant.id,
    merchantName: merchant.name,
    groupName: merchant.groupName,
    status: 'pending',
    requestedAt: now,
    updatedAt: now,
  });
}

// 取所有 pending，标记 processing 后返回（采集端拉取）。
export function claimPending(): ScreenshotRequest[] {
  const now = Date.now();
  sweep(now);
  const map = store();
  const claimed: ScreenshotRequest[] = [];
  for (const req of map.values()) {
    if (req.status === 'pending') {
      req.status = 'processing';
      req.updatedAt = now;
      claimed.push({ ...req });
    }
  }
  return claimed;
}

// 采集端回报发送成功，移除该条。返回是否命中。
export function complete(groupName: string): boolean {
  sweep(Date.now());
  return store().delete(groupName);
}
```

```ts
import type { NextApiRequest, NextApiResponse } from 'next';
import db from '@/lib/sqlite-db';
import { Merchant } from '@/lib/types';
import { enqueue } from '@/lib/screenshot-request-cache';

// WorkTool 机器人消息回调。群里发「机器人发图」→ 记录该群所属商家的发图请求。
// 契约：https://doc.worktool.ymdyes.cn/doc-861677
// - 必须返回 { code: 0, message: 'success' }；不支持在响应内下发指令。

const TRIGGER_KEYWORD = '机器人发图';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') {
    res.setHeader('Allow', ['POST']);
    return res.status(405).json({ code: 1, message: 'Method Not Allowed' });
  }

  // 鉴权：URL token（WorkTool 回调不会带 x-api-key）
  const token = req.query.token;
  const expected = process.env.WORKTOOL_CALLBACK_TOKEN;
  if (!expected || token !== expected) {
    return res.status(401).json({ code: 1, message: 'invalid token' });
  }

  try {
    const body = req.body || {};
    const roomType = Number(body.roomType);
    const textType = Number(body.textType);
    const spoken = String(body.spoken ?? '').trim();
    const groupName = String(body.groupName ?? '').trim();

    // 过滤：仅群(1/3) + 文本(1) + 精确触发词
    const isGroup = roomType === 1 || roomType === 3;
    if (!isGroup || textType !== 1 || spoken !== TRIGGER_KEYWORD) {
      return res.status(200).json({ code: 0, message: 'success' });
    }

    // 匹配商家
    await db.init();
    const merchants = await db.getAllMerchants();
    const merchant = merchants.find((m: Merchant) => m.groupName === groupName);

    if (!merchant) {
      console.warn(`[worktool-callback] 未匹配到商家，群名: ${groupName}`);
      return res.status(200).json({ code: 0, message: 'success' });
    }

    enqueue(merchant);
    console.log(`[worktool-callback] 已入队发图请求：${merchant.name} / ${groupName}`);
    return res.status(200).json({ code: 0, message: 'success' });
  } catch (error) {
    console.error('[worktool-callback] 处理回调出错:', error);
    // 仍返回成功，避免 WorkTool 重试风暴；错误已记日志
    return res.status(200).json({ code: 0, message: 'success' });
  }
}
```

- [ ] **Step 1: 写轮询/回报接口**

Create `xc_store/src/pages/api/public/screenshot-requests.ts`:

```ts
import type { NextApiRequest, NextApiResponse } from 'next';
import db from '@/lib/sqlite-db';
import { claimPending, complete } from '@/lib/screenshot-request-cache';

// 采集端接口：
// GET  拉取待发图请求（pending → processing）
// POST 回报发送完成（按 groupName 移除）
export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'GET' && req.method !== 'POST') {
    res.setHeader('Allow', ['GET', 'POST']);
    return res.status(405).json({ success: false, error: '不支持的请求方法' });
  }

  try {
    await db.init();

    // API Key 校验（沿用现有 public 范式）
    const apiKey = req.headers['x-api-key'] || req.query.apiKey;
    const systemApiKey = await db.getSetting('apiKey');
    if (!apiKey || !systemApiKey || apiKey !== systemApiKey) {
      return res.status(401).json({ success: false, error: '无效的API密钥' });
    }

    if (req.method === 'GET') {
      const data = claimPending();
      return res.status(200).json({ success: true, data, total: data.length });
    }

    // POST：回报完成
    const groupName = String(req.body?.groupName ?? '').trim();
    if (!groupName) {
      return res.status(400).json({ success: false, error: '缺少 groupName' });
    }
    const removed = complete(groupName);
    return res.status(200).json({ success: true, removed });
  } catch (error) {
    console.error('[screenshot-requests] 处理出错:', error);
    return res.status(500).json({
      success: false,
      error: '服务器内部错误',
      message: error instanceof Error ? error.message : '未知错误',
    });
  }
}
```

```python
    async def _poll_and_handle_on_demand_screenshots(self):
        """
        按需发图：轮询服务端待发图请求，对匹配账号截图并发回对应群，成功后回报完成。
        运行在商家操作线程的事件循环里，单条任务用超时包裹，单条卡死不拖垮主循环。
        """
        api_url = self.local_data.get('api_url', '').rstrip('/')
        api_key = self.local_data.get('api_key', '')
        if not api_url or not api_key:
            return

        endpoint = build_api_url(api_url, ApiEndpoints.SCREENSHOT_REQUESTS)
        headers = {"x-api-key": api_key, "Content-Type": "application/json"}

        # 1) 拉取待发图请求
        try:
            async with AiohttpUtils(timeout=15, headers=headers) as client:
                resp = await client.get(endpoint)
        except Exception as e:
            logger.warning(f"[按需发图] 拉取请求失败: {e}")
            return

        if not resp or not resp.get("success"):
            return
        requests_list = resp.get("data", []) or []
        if not requests_list:
            return

        logger.info(f"[按需发图] 拉到 {len(requests_list)} 条待发图请求")

        # 2) 逐条处理
        for item in requests_list:
            if not os.path.exists(TASK_RUNNING_FLAG):
                break
            merchant_id = str(item.get("merchantId", ""))
            group_name = item.get("groupName", "")

            account_data = next(
                (a for a in self.accounts_data if str(a.get("api_id", "")) == merchant_id),
                None
            )
            if not account_data:
                logger.warning(f"[按需发图] 未找到匹配账号(merchantId={merchant_id}, 群={group_name})，跳过，不回报")
                continue
            if self.browser is None:
                logger.warning("[按需发图] 浏览器不可用，跳过本轮")
                break

            username = account_data.get('账号', '')
            context = None
            try:
                # 复用现有单账号上下文创建方式
                cookie_path = os.path.join(COOKIE_PATH, f"{username}.json")
                if cookie_path and os.path.exists(cookie_path):
                    context = await self.browser.new_context(storage_state=cookie_path, no_viewport=True)
                else:
                    context = await self.browser.new_context(no_viewport=True)
                page = await context.new_page()
                stealth = Stealth()
                await stealth.apply_stealth_async(page)

                # 复用 process_page 的按需发图分支，单条超时包裹
                await asyncio.wait_for(
                    self.process_page(page, 0, account_data, task_type="on_demand_screenshot"),
                    timeout=ON_DEMAND_JOB_TIMEOUT
                )

                # 3) 回报完成
                async with AiohttpUtils(timeout=15, headers=headers) as client:
                    await client.post(endpoint, json_data={"groupName": group_name})
                logger.info(f"[按需发图] 已发图并回报完成：{account_data.get('api_name', '')} / {group_name}")
            except asyncio.TimeoutError:
                logger.error(f"[按需发图] 单条任务超时({ON_DEMAND_JOB_TIMEOUT}s)，放弃并不回报：群={group_name}")
            except Exception as e:
                logger.error(f"[按需发图] 处理失败,不回报(留待服务端 TTL 重试)：群={group_name} - {e}", exc_info=True)
            finally:
                if context is not None:
                    try:
                        await context.close()
                    except Exception:
                        pass
```
